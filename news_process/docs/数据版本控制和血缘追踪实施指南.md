## 背景

当前项目的中间数据存储在 `data/processed/*.parquet` 文件中，但缺少以下关键信息：
- 数据是由哪个版本的代码生成的？
- 使用了哪个版本的配置文件？
- 依赖的输入数据是什么？
- 何时生成的？

这导致无法追溯研究结果的生成过程，影响科研可重现性。

---

## 目标

1. **数据版本标记**：在 parquet 文件元数据中记录生成信息
2. **血缘追踪**：记录数据依赖关系
3. **完整性验证**：验证数据未被篡改
4. **可视化**：生成数据血缘关系图

### 步骤 1: 创建数据版本控制模块（90-120 分钟）

创建 `news_process/core/data_versioning.py`：

```python
"""news_process.core.data_versioning — 数据版本控制和血缘追踪。

在 parquet 文件元数据中记录：
- Git commit hash
- 配置文件 hash
- 生成时间戳
- 依赖的输入文件 hash
- Python 版本和关键依赖版本
"""

from __future__ import annotations

import hashlib
import json
import logging
import subprocess
from datetime import datetime
from pathlib import Path
from typing import Optional

import pandas as pd
import pyarrow as pa
import pyarrow.parquet as pq

logger = logging.getLogger(__name__)


def get_git_commit_hash() -> str:
    """获取当前 Git commit hash。"""
    try:
        result = subprocess.run(
            ["git", "rev-parse", "HEAD"],
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError:
        logger.warning("无法获取 Git commit hash，可能不在 Git 仓库中")
        return "unknown"


def get_git_branch() -> str:
    """获取当前 Git 分支名。"""
    try:
        result = subprocess.run(
            ["git", "rev-parse", "--abbrev-ref", "HEAD"],
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError:
        return "unknown"


def compute_file_hash(file_path: str | Path) -> str:
    """计算文件的 SHA256 hash。"""
    sha256 = hashlib.sha256()
    with open(file_path, "rb") as f:
        for chunk in iter(lambda: f.read(8192), b""):
            sha256.update(chunk)
    return sha256.hexdigest()


def compute_config_hash(config: dict) -> str:
    """计算配置字典的 hash。"""
    # 将配置序列化为 JSON 字符串（排序键以确保一致性）
    config_str = json.dumps(config, sort_keys=True, ensure_ascii=False)
    return hashlib.sha256(config_str.encode()).hexdigest()


def get_python_version() -> str:
    """获取 Python 版本。"""
    import sys
    return f"{sys.version_info.major}.{sys.version_info.minor}.{sys.version_info.micro}"


def get_key_dependencies() -> dict[str, str]:
    """获取关键依赖的版本。"""
    import pandas
    import numpy
    import torch
    import transformers
    
    return {
        "pandas": pandas.__version__,
        "numpy": numpy.__version__,
        "torch": torch.__version__,
        "transformers": transformers.__version__,
    }


class DataVersionMetadata:
    """数据版本元数据。"""
    
    def __init__(
        self,
        stage_name: str,
        config: dict,
        input_files: Optional[list[str]] = None,
    ):
        self.stage_name = stage_name
        self.config = config
        self.input_files = input_files or []
        
        # 收集版本信息
        self.git_commit = get_git_commit_hash()
        self.git_branch = get_git_branch()
        self.config_hash = compute_config_hash(config)
        self.timestamp = datetime.now().isoformat()
        self.python_version = get_python_version()
        self.dependencies = get_key_dependencies()
        
        # 计算输入文件 hash
        self.input_hashes = {}
        for input_file in self.input_files:
            if Path(input_file).exists():
                self.input_hashes[input_file] = compute_file_hash(input_file)
    
    def to_dict(self) -> dict:
        """转换为字典。"""
        return {
            "stage_name": self.stage_name,
            "git_commit": self.git_commit,
            "git_branch": self.git_branch,
            "config_hash": self.config_hash,
            "timestamp": self.timestamp,
            "python_version": self.python_version,
            "dependencies": self.dependencies,
            "input_files": self.input_files,
            "input_hashes": self.input_hashes,
        }
    
    def to_json(self) -> str:
        """转换为 JSON 字符串。"""
        return json.dumps(self.to_dict(), indent=2, ensure_ascii=False)


def save_parquet_with_metadata(
    df: pd.DataFrame,
    output_path: str | Path,
    stage_name: str,
    config: dict,
    input_files: Optional[list[str]] = None,
):
    """
    保存 DataFrame 到 parquet 文件，并附加版本元数据。
    
    Args:
        df: 要保存的 DataFrame
        output_path: 输出文件路径
        stage_name: 阶段名称
        config: 配置字典
        input_files: 依赖的输入文件列表
    """
    # 创建版本元数据
    metadata = DataVersionMetadata(stage_name, config, input_files)
    
    # 转换为 PyArrow Table
    table = pa.Table.from_pandas(df)
    
    # 添加自定义元数据
    existing_metadata = table.schema.metadata or {}
    custom_metadata = {
        b"data_version": metadata.to_json().encode("utf-8"),
    }
    merged_metadata = {**existing_metadata, **custom_metadata}
    
    # 创建新的 schema with metadata
    new_schema = table.schema.with_metadata(merged_metadata)
    table = table.cast(new_schema)
    
    # 保存到文件
    pq.write_table(table, output_path)
    
    logger.info(
        f"已保存 {output_path} with metadata:\n"
        f"  Git commit: {metadata.git_commit[:8]}\n"
        f"  Config hash: {metadata.config_hash[:8]}\n"
        f"  Timestamp: {metadata.timestamp}"
    )


def read_parquet_metadata(file_path: str | Path) -> Optional[dict]:
    """
    读取 parquet 文件的版本元数据。
    
    Args:
        file_path: parquet 文件路径
    
    Returns:
        版本元数据字典，如果不存在则返回 None
    """
    try:
        parquet_file = pq.ParquetFile(file_path)
        metadata = parquet_file.schema_arrow.metadata
        
        if metadata and b"data_version" in metadata:
            version_json = metadata[b"data_version"].decode("utf-8")
            return json.loads(version_json)
        
        return None
    except Exception as e:
        logger.warning(f"无法读取 {file_path} 的元数据: {e}")
        return None


def verify_data_integrity(file_path: str | Path) -> bool:
    """
    验证数据完整性。
    
    检查：
    1. 文件是否存在
    2. 元数据是否存在
    3. 输入文件 hash 是否匹配（如果输入文件仍存在）
    
    Args:
        file_path: parquet 文件路径
    
    Returns:
        True 如果数据完整，False 如果存在问题
    """
    if not Path(file_path).exists():
        logger.error(f"文件不存在: {file_path}")
        return False
    
    metadata = read_parquet_metadata(file_path)
    if metadata is None:
        logger.warning(f"文件缺少版本元数据: {file_path}")
        return False
    
    # 验证输入文件 hash
    for input_file, expected_hash in metadata.get("input_hashes", {}).items():
        if Path(input_file).exists():
            actual_hash = compute_file_hash(input_file)
            if actual_hash != expected_hash:
                logger.error(
                    f"输入文件 hash 不匹配: {input_file}\n"
                    f"  期望: {expected_hash[:8]}...\n"
                    f"  实际: {actual_hash[:8]}..."
                )
                return False
    
    logger.info(f"数据完整性验证通过: {file_path}")
    return True
```

### 步骤 3: 创建血缘追踪可视化（90-120 分钟）

创建 `scripts/visualize_lineage.py`：

```python
"""生成数据血缘关系图。"""

import argparse
from pathlib import Path
from typing import Dict, List, Set

from news_process.core.data_versioning import read_parquet_metadata
from news_process.core.schemas import OUTPUT_REGISTRY


def build_lineage_graph() -> Dict[str, List[str]]:
    """
    构建数据血缘关系图。
    
    Returns:
        字典，键为输出文件，值为输入文件列表
    """
    lineage = {}
    
    for stage, output_files in OUTPUT_REGISTRY.items():
        for output_file in output_files:
            output_path = Path("data/processed") / output_file
            
            if output_path.exists():
                metadata = read_parquet_metadata(output_path)
                if metadata:
                    input_files = metadata.get("input_files", [])
                    lineage[output_file] = input_files
    
    return lineage


def generate_mermaid_diagram(lineage: Dict[str, List[str]]) -> str:
    """
    生成 Mermaid 格式的血缘关系图。
    
    Args:
        lineage: 血缘关系字典
    
    Returns:
        Mermaid 图表字符串
    """
    lines = ["graph TD"]
    
    # 添加节点和边
    for output_file, input_files in lineage.items():
        output_node = output_file.replace(".", "_")
        
        for input_file in input_files:
            input_node = Path(input_file).name.replace(".", "_")
            lines.append(f"    {input_node}[{Path(input_file).name}] --> {output_node}[{output_file}]")
    
    return "\n".join(lines)


def generate_dot_diagram(lineage: Dict[str, List[str]]) -> str:
    """
    生成 Graphviz DOT 格式的血缘关系图。
    
    Args:
        lineage: 血缘关系字典
    
    Returns:
        DOT 图表字符串
    """
    lines = ["digraph lineage {", "    rankdir=LR;", "    node [shape=box];"]
    
    # 收集所有节点
    all_nodes: Set[str] = set()
    for output_file, input_files in lineage.items():
        all_nodes.add(output_file)
        all_nodes.update(Path(f).name for f in input_files)
    
    # 添加节点定义
    for node in sorted(all_nodes):
        node_id = node.replace(".", "_")
        lines.append(f'    {node_id} [label="{node}"];')
    
    # 添加边
    for output_file, input_files in lineage.items():
        output_id = output_file.replace(".", "_")
        for input_file in input_files:
            input_id = Path(input_file).name.replace(".", "_")
            lines.append(f"    {input_id} -> {output_id};")
    
    lines.append("}")
    return "\n".join(lines)


def main():
    parser = argparse.ArgumentParser(description="生成数据血缘关系图")
    parser.add_argument(
        "--format",
        choices=["mermaid", "dot"],
        default="mermaid",
        help="输出格式（mermaid 或 dot）"
    )
    parser.add_argument(
        "--output",
        default="data_lineage.md",
        help="输出文件路径"
    )
    
    args = parser.parse_args()
    
    # 构建血缘关系图
    lineage = build_lineage_graph()
    
    # 生成图表
    if args.format == "mermaid":
        diagram = generate_mermaid_diagram(lineage)
        content = f"# 数据血缘关系图\n\n```mermaid\n{diagram}\n```\n"
    else:
        diagram = generate_dot_diagram(lineage)
        content = diagram
    
    # 保存到文件
    with open(args.output, "w", encoding="utf-8") as f:
        f.write(content)
    
    print(f"✓ 血缘关系图已保存到: {args.output}")
    print(f"  包含 {len(lineage)} 个数据文件")


if __name__ == "__main__":
    main()
```

### 步骤 4: 创建数据审计工具（60 分钟）

```python
"""数据版本审计工具。"""

import argparse
from pathlib import Path
from typing import List

from news_process.core.data_versioning import read_parquet_metadata, verify_data_integrity
from news_process.core.schemas import OUTPUT_REGISTRY


def audit_all_data() -> List[dict]:
    """
    审计所有数据文件。
    
    Returns:
        审计结果列表
    """
    results = []
    
    for stage, output_files in OUTPUT_REGISTRY.items():
        for output_file in output_files:
            output_path = Path("data/processed") / output_file
            
            result = {
                "file": output_file,
                "exists": output_path.exists(),
                "has_metadata": False,
                "integrity_ok": False,
                "metadata": None,
            }
            
            if output_path.exists():
                metadata = read_parquet_metadata(output_path)
                result["has_metadata"] = metadata is not None
                result["metadata"] = metadata
                
                if metadata:
                    result["integrity_ok"] = verify_data_integrity(output_path)
            
            results.append(result)
    
    return results


def print_audit_report(results: List[dict]):
    """打印审计报告。"""
    print("=" * 80)
    print("数据版本审计报告")
    print("=" * 80)
    print()
    
    total = len(results)
    exists = sum(1 for r in results if r["exists"])
    has_metadata = sum(1 for r in results if r["has_metadata"])
    integrity_ok = sum(1 for r in results if r["integrity_ok"])
    
    print(f"总文件数: {total}")
    print(f"存在的文件: {exists} ({exists/total*100:.1f}%)")
    print(f"包含元数据: {has_metadata} ({has_metadata/total*100:.1f}%)")
    print(f"完整性验证通过: {integrity_ok} ({integrity_ok/total*100:.1f}%)")
    print()
    
    # 详细信息
    print("详细信息:")
    print("-" * 80)
    
    for result in results:
        status = "✓" if result["integrity_ok"] else "✗"
        print(f"{status} {result['file']}")
        
        if result["metadata"]:
            metadata = result["metadata"]
            print(f"    Git commit: {metadata.get('git_commit', 'N/A')[:8]}")
            print(f"    Timestamp: {metadata.get('timestamp', 'N/A')}")
            print(f"    Config hash: {metadata.get('config_hash', 'N/A')[:8]}")
        elif result["exists"]:
            print("    ⚠ 缺少版本元数据")
        else:
            print("    ⚠ 文件不存在")
        
        print()


def main():
    parser = argparse.ArgumentParser(description="审计数据版本和完整性")
    parser.add_argument(
        "--json",
        action="store_true",
        help="以 JSON 格式输出结果"
    )
    
    args = parser.parse_args()
    
    results = audit_all_data()
    
    if args.json:
        import json
        print(json.dumps(results, indent=2, ensure_ascii=False))
    else:
        print_audit_report(results)


if __name__ == "__main__":
    main()
```

import pandas as pd

## 注意事项

1. **元数据大小**：版本元数据存储在 parquet 文件的 schema metadata 中，不会显著增加文件大小
2. **向后兼容**：读取元数据的函数会优雅地处理没有元数据的旧文件
3. **Git 依赖**：需要在 Git 仓库中运行才能获取 commit hash
4. **输入文件路径**：使用相对于项目根的路径，确保可移植性
5. **性能影响**：计算文件 hash 会增加少量开销，但对于大型数据集影响可忽略
