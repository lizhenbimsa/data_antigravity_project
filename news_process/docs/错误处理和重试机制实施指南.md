# 错误处理和重试机制实施指南

**优先级**: P1 (高优先级)  
**预计工作量**: 8-12 小时  
**技术风险**: 中等  
**目标版本**: v2.2.0

---

## 背景

当前流水线在遇到临时性错误（如网络问题、API 限流）时会直接失败，需要手动重新运行整个流水线。这对于长时间运行的流水线（特别是 LLM API 调用密集的阶段）造成了显著的时间和资源浪费。

---

## 目标

1. **API 调用重试**：为 LLM 和 Embedding API 调用添加指数退避重试
2. **阶段级断点续传**：支持从失败的阶段继续执行
3. **错误分类和聚合**：区分临时性错误和永久性错误
4. **友好的错误恢复指引**：提供清晰的错误信息和恢复建议

### 步骤 2: 创建重试装饰器模块（60 分钟）

创建 `news_process/core/retry.py`：

```python
"""news_process.core.retry — API 调用重试装饰器。

提供统一的重试策略，用于处理临时性错误（网络问题、API 限流等）。
"""

from __future__ import annotations

import logging
from typing import Callable, Type

from tenacity import (
    retry,
    stop_after_attempt,
    wait_exponential,
    retry_if_exception_type,
    before_sleep_log,
    after_log,
)
import aiohttp

logger = logging.getLogger(__name__)


# 临时性错误类型（应该重试）
TRANSIENT_ERRORS = (
    aiohttp.ClientError,
    aiohttp.ServerTimeoutError,
    ConnectionError,
    TimeoutError,
)


def retry_on_transient_error(
    max_attempts: int = 5,
    min_wait: float = 1.0,
    max_wait: float = 60.0,
    multiplier: float = 2.0,
) -> Callable:
    """
    为函数添加指数退避重试机制。
    
    Args:
        max_attempts: 最大重试次数（包括首次尝试）
        min_wait: 最小等待时间（秒）
        max_wait: 最大等待时间（秒）
        multiplier: 等待时间倍增因子
    
    Returns:
        装饰器函数
    
    Example:
        @retry_on_transient_error(max_attempts=3)
        async def call_api(url: str) -> dict:
            async with aiohttp.ClientSession() as session:
                async with session.get(url) as response:
                    return await response.json()
    """
    return retry(
        stop=stop_after_attempt(max_attempts),
        wait=wait_exponential(
            multiplier=multiplier,
            min=min_wait,
            max=max_wait,
        ),
        retry=retry_if_exception_type(TRANSIENT_ERRORS),
        before_sleep=before_sleep_log(logger, logging.WARNING),
        after=after_log(logger, logging.INFO),
        reraise=True,
    )


def is_transient_error(exc: Exception) -> bool:
    """
    判断异常是否为临时性错误（可重试）。
    
    Args:
        exc: 异常对象
    
    Returns:
        True 如果是临时性错误，False 如果是永久性错误
    """
    return isinstance(exc, TRANSIENT_ERRORS)


def classify_error(exc: Exception) -> dict:
    """
    对错误进行分类，返回错误信息和恢复建议。
    
    Args:
        exc: 异常对象
    
    Returns:
        包含错误分类、消息和恢复建议的字典
    """
    error_info = {
        "type": type(exc).__name__,
        "message": str(exc),
        "is_transient": is_transient_error(exc),
        "recovery_suggestion": "",
    }
    
    if isinstance(exc, aiohttp.ClientError):
        error_info["category"] = "network"
        error_info["recovery_suggestion"] = (
            "网络连接问题。请检查网络连接后重试。"
        )
    elif isinstance(exc, TimeoutError):
        error_info["category"] = "timeout"
        error_info["recovery_suggestion"] = (
            "请求超时。可能是网络延迟或服务器负载过高，请稍后重试。"
        )
    elif "rate limit" in str(exc).lower():
        error_info["category"] = "rate_limit"
        error_info["recovery_suggestion"] = (
            "API 限流。请等待一段时间后重试，或检查 API 配额。"
        )
    elif "authentication" in str(exc).lower() or "unauthorized" in str(exc).lower():
        error_info["category"] = "auth"
        error_info["is_transient"] = False
        error_info["recovery_suggestion"] = (
            "认证失败。请检查 API 密钥是否正确配置（.env 文件中的 YUNWU_API_KEY）。"
        )
    else:
        error_info["category"] = "unknown"
        error_info["recovery_suggestion"] = (
            "未知错误。请查看完整错误日志以获取更多信息。"
        )
    
    return error_info
```

                return await response.json()

            f"错误类型: {error_info['category']}\n"
            f"恢复建议: {error_info['recovery_suggestion']}"

#### 4.1 创建进度追踪模块

创建 `news_process/core/checkpoint.py`：

```python
"""news_process.core.checkpoint — 阶段级断点续传。

记录流水线执行进度，支持从失败的阶段继续执行。
"""

from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Optional
from datetime import datetime

logger = logging.getLogger(__name__)


class CheckpointManager:
    """流水线检查点管理器。"""
    
    def __init__(self, run_id: str, checkpoint_dir: str = "logs/checkpoints"):
        self.run_id = run_id
        self.checkpoint_dir = Path(checkpoint_dir)
        self.checkpoint_dir.mkdir(parents=True, exist_ok=True)
        self.checkpoint_file = self.checkpoint_dir / f"{run_id}.json"
    
    def save_stage_completion(self, stage_name: str, success: bool, error: Optional[str] = None):
        """记录阶段完成状态。"""
        checkpoint_data = self._load_checkpoint()
        
        checkpoint_data["stages"][stage_name] = {
            "success": success,
            "timestamp": datetime.now().isoformat(),
            "error": error,
        }
        
        self._save_checkpoint(checkpoint_data)
        logger.info(f"检查点已保存: {stage_name} ({'成功' if success else '失败'})")
    
    def is_stage_completed(self, stage_name: str) -> bool:
        """检查阶段是否已完成。"""
        checkpoint_data = self._load_checkpoint()
        stage_info = checkpoint_data["stages"].get(stage_name)
        return stage_info is not None and stage_info["success"]
    
    def get_failed_stages(self) -> list[str]:
        """获取所有失败的阶段。"""
        checkpoint_data = self._load_checkpoint()
        return [
            name for name, info in checkpoint_data["stages"].items()
            if not info["success"]
        ]
    
    def clear(self):
        """清除检查点（新的流水线运行）。"""
        if self.checkpoint_file.exists():
            self.checkpoint_file.unlink()
            logger.info(f"检查点已清除: {self.run_id}")
    
    def _load_checkpoint(self) -> dict:
        """加载检查点数据。"""
        if self.checkpoint_file.exists():
            with open(self.checkpoint_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return {"run_id": self.run_id, "stages": {}}
    
    def _save_checkpoint(self, data: dict):
        """保存检查点数据。"""
        with open(self.checkpoint_file, 'w', encoding='utf-8') as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
```

from news_process.core.checkpoint import CheckpointManager

def run_pipeline(config: dict, resume: bool = False):
    """
    运行流水线。
    
    Args:
        config: 配置字典
        resume: 是否从上次失败处继续（断点续传）
    """
    run_id = generate_run_id()
    checkpoint_mgr = CheckpointManager(run_id)
    
    if not resume:
        checkpoint_mgr.clear()
    
    for stage in STAGES_MAIN:
        # 如果是续传模式且阶段已完成，跳过
        if resume and checkpoint_mgr.is_stage_completed(stage.name):
            logger.info(f"跳过已完成的阶段: {stage.name}")
            continue
        
        try:
            logger.info(f"开始执行阶段: {stage.name}")
            run_stage(stage, config)
            checkpoint_mgr.save_stage_completion(stage.name, success=True)
        except Exception as e:
            error_msg = str(e)
            checkpoint_mgr.save_stage_completion(stage.name, success=False, error=error_msg)
            logger.error(f"阶段失败: {stage.name}\n错误: {error_msg}")
            
            # 显示恢复指引
            print(f"\n{'='*70}")
            print(f"流水线在阶段 '{stage.name}' 失败")
            print(f"错误: {error_msg}")
            print(f"\n恢复选项:")
            print(f"1. 修复问题后，使用 --resume 参数继续:")
            print(f"   python run_pipeline_function.py --resume")
            print(f"2. 从头开始重新运行:")
            print(f"   python run_pipeline_function.py")
            print(f"{'='*70}\n")
            
            raise
```

import argparse

def main():
    parser = argparse.ArgumentParser(description="运行 news_process 流水线")

        "--resume",

        help="从上次失败处继续执行（断点续传）"

        action="store_true",

    config = load_config()
    
    if args.clear_checkpoints:
        # 清除所有检查点

        if checkpoint_dir.exists():
            for checkpoint_file in checkpoint_dir.glob("*.json"):
                checkpoint_file.unlink()
            print("✓ 所有检查点已清除")
    
    run_pipeline(config, resume=args.resume)

if __name__ == "__main__":
    main()

from news_process.core.retry import (
    retry_on_transient_error,

import aiohttp


def test_is_transient_error():
    """测试临时性错误识别"""
    assert is_transient_error(aiohttp.ClientError())
    assert is_transient_error(TimeoutError())
    assert not is_transient_error(ValueError())


def test_classify_error_network():
    """测试网络错误分类"""
    error = aiohttp.ClientError("Connection failed")
    info = classify_error(error)
    
    assert info["category"] == "network"
    assert info["is_transient"] is True
    assert "网络连接" in info["recovery_suggestion"]


def test_classify_error_auth():
    """测试认证错误分类"""
    error = Exception("Authentication failed")
    info = classify_error(error)
    
    assert info["category"] == "auth"
    assert info["is_transient"] is False
    assert "API 密钥" in info["recovery_suggestion"]


@pytest.mark.asyncio
async def test_retry_decorator_success():
    """测试重试装饰器 - 成功场景"""
    call_count = 0
    
    @retry_on_transient_error(max_attempts=3)
    async def flaky_function():
        nonlocal call_count
        call_count += 1
        if call_count < 2:
            raise aiohttp.ClientError("Temporary failure")
        return "success"
    
    result = await flaky_function()
    assert result == "success"
    assert call_count == 2  # 第一次失败，第二次成功


@pytest.mark.asyncio
async def test_retry_decorator_max_attempts():
    """测试重试装饰器 - 达到最大重试次数"""
    call_count = 0
    
    @retry_on_transient_error(max_attempts=3)
    async def always_fail():
        nonlocal call_count
        call_count += 1
        raise aiohttp.ClientError("Persistent failure")
    
    with pytest.raises(aiohttp.ClientError):
        await always_fail()
    
    assert call_count == 3  # 尝试了 3 次
```
