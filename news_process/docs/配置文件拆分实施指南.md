### 步骤 4: 验证配置等价性（30 分钟）

#### 4.1 创建验证脚本

```python
# scripts/validate_config_split.py
from news_process.core.config import load_config
import yaml

# 加载拆分后的配置
split_config = load_config()

# 加载备份的单体配置
with open('config/config.yaml.backup', 'r', encoding='utf-8') as f:
    monolithic_config = yaml.safe_load(f)

# 比较两者是否等价
def compare_configs(a, b, path=""):
    if type(a) != type(b):
        print(f"❌ 类型不匹配 {path}: {type(a)} vs {type(b)}")
        return False
    
    if isinstance(a, dict):
        if set(a.keys()) != set(b.keys()):
            print(f"❌ 键不匹配 {path}: {set(a.keys()) ^ set(b.keys())}")
            return False
        for key in a:
            if not compare_configs(a[key], b[key], f"{path}.{key}"):
                return False
    elif isinstance(a, list):
        if len(a) != len(b):
            print(f"❌ 列表长度不匹配 {path}: {len(a)} vs {len(b)}")
            return False
        for i, (item_a, item_b) in enumerate(zip(a, b)):
            if not compare_configs(item_a, item_b, f"{path}[{i}]"):
                return False
    else:
        if a != b:
            print(f"❌ 值不匹配 {path}: {a} vs {b}")
            return False
    
    return True

if compare_configs(split_config, monolithic_config):
    print("✅ 配置拆分验证通过：拆分后的配置与原配置完全等价")
else:
    print("❌ 配置拆分验证失败：存在差异")
    exit(1)
```

```python
import pytest
from news_process.core.config import load_config

def test_config_split_mode_detected():
    """验证拆分模式被正确检测"""
    # 这个测试会在拆分文件存在时通过
    config = load_config()
    assert config is not None
    assert isinstance(config, dict)

def test_all_required_sections_present():
    """验证所有必需的配置段都存在"""
    config = load_config()
    
    required_sections = [
        'paths', 'filenames', 'parallel',
        'preprocessing', 'normalization', 'deduplication',
        'bank_news_classification', 'fintech_news_classification',
        'tone_llm_processing', 'sentiment_analysis',
        'embedding_api', 'panel_building', 'validation'
    ]
    
    for section in required_sections:
        assert section in config, f"缺少配置段: {section}"

def test_prompt_templates_loaded():
    """验证 prompt 模板被正确加载"""
    config = load_config()
    
    # 验证 prompt 模板已从文件加载到配置中
    assert 'prompt_template' in config['tone_llm_processing']
    assert 'title_sentiment_prompt_template' in config['tone_llm_processing']
    
    # 验证模板内容不为空
    assert len(config['tone_llm_processing']['prompt_template']) > 100
    assert len(config['tone_llm_processing']['title_sentiment_prompt_template']) > 50
```
